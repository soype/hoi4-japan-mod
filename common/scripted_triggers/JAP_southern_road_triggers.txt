# A country that Japan should treat as a Tripartite Pact partner for this chain.
# Germany and Italy are the core signatories; their puppets and other Axis members
# are covered as well. The pact is considered active if Japan has signed it, or if
# Japan has somehow joined the Axis directly.
JAP_sr_is_tripartite_signer = {
	JAP = {
		OR = {
			has_completed_focus = JAP_sea_tripartite_pact
			AND = {
				country_exists = GER
				is_in_faction_with = GER
			}
			AND = {
				country_exists = ITA
				is_in_faction_with = ITA
			}
		}
	}
	NOT = { has_war_with = JAP }
	OR = {
		AND = {
			tag = GER
			has_government = fascism
		}
		AND = {
			tag = ITA
			has_government = fascism
		}
		AND = {
			country_exists = GER
			is_in_faction_with = GER
			GER = { has_government = fascism }
		}
		AND = {
			country_exists = ITA
			is_in_faction_with = ITA
			ITA = { has_government = fascism }
		}
		AND = {
			country_exists = GER
			is_subject_of = GER
			GER = { has_government = fascism }
		}
		AND = {
			country_exists = ITA
			is_subject_of = ITA
			ITA = { has_government = fascism }
		}
	}
}

# True if the country in scope, or one of its direct subjects, controls any target state.
JAP_sr_controls_target_states = {
	OR = {
		any_controlled_state = {
			JAP_sr_target_state = yes
		}
		any_country = {
			is_subject_of = PREV
			any_controlled_state = {
				JAP_sr_target_state = yes
			}
		}
	}
}

# Japan's AI uses this to decide whether to pay the Tripartite partner's price.
# Japan only accepts when the Axis is at war with the USSR AND the USSR is losing
# or the Axis is clearly stronger. Otherwise Tokyo sees weakness and refuses.
JAP_sr_should_accept_axis_demands = {
	# Is the Axis at war with the USSR (or has the USSR already been defeated)?
	OR = {
		NOT = { country_exists = SOV }
		AND = {
			country_exists = SOV
			OR = {
				AND = {
					country_exists = GER
					SOV = { has_war_with = GER }
				}
				AND = {
					country_exists = ITA
					SOV = { has_war_with = ITA }
				}
			}
		}
	}
	# Is the USSR losing, or are the Axis forces too great?
	OR = {
		NOT = { country_exists = SOV }
		AND = {
			country_exists = SOV
			SOV = { surrender_progress > 0.5 }
		}
		AND = {
			country_exists = GER
			GER = {
				strength_ratio = {
					tag = JAP
					ratio > 1.5
				}
			}
		}
		AND = {
			country_exists = ITA
			ITA = {
				strength_ratio = {
					tag = JAP
					ratio > 1.5
				}
			}
		}
		AND = {
			country_exists = FROM
			FROM = {
				strength_ratio = {
					tag = JAP
					ratio > 1.5
				}
			}
		}
	}
}
